cmake_minimum_required(VERSION 3.24)
project(CyCodeIDE VERSION 0.2.3 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(MSVC)
  add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/EHsc>" /utf-8)
endif()
get_filename_component(CYCODE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(CYCODE_CORE_ROOT "" CACHE PATH "Explicit CyCode native source checkout")
if(NOT EXISTS "${CYCODE_CORE_ROOT}/CMakeLists.txt" OR NOT EXISTS "${CYCODE_CORE_ROOT}/common/api-version.txt")
  message(FATAL_ERROR "Resolve the pinned Core with scripts/build.py or set CYCODE_CORE_ROOT")
endif()
set(CYCODE_DEPS "${CYCODE_ROOT}/runtime/deps" CACHE PATH "Build dependency sources")
option(BUILD_TESTING "Build opt-in native tests; never run automatically" OFF)
include(CTest)
find_package(Threads REQUIRED)
set(CYCODE_BUILD_CLI ON CACHE BOOL "Native service for the IDE" FORCE)
set(CYCODE_BUILD_UPDATES ON CACHE BOOL "Installation services" FORCE)
set(CYCODE_BUILD_PROVIDERS ON CACHE BOOL "Bundled native providers" FORCE)
set(CYCODE_BUILD_GUI OFF CACHE BOOL "GUI is owned by this repository" FORCE)
add_subdirectory("${CYCODE_CORE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/core")
set(CYCODE_EDITOR_ROOT "${CYCODE_ROOT}/runtime/deps/cycode-reader" CACHE PATH "Shared graphical editor API 1")
set(CYUI_DIR "" CACHE PATH "Optional clean CyUI source at the locked revision")
set(CYCODE_EDITOR_DEPS "${CYCODE_DEPS}")
set(CYCODE_EDITOR_BUILD_GUI ON)
# Reader API 1 must consume supplied native targets instead of defining duplicates.
file(READ "${CYCODE_EDITOR_ROOT}/editor/api-version.txt" CYCODE_EDITOR_API)
string(STRIP "${CYCODE_EDITOR_API}" CYCODE_EDITOR_API)
if(NOT CYCODE_EDITOR_API STREQUAL "1")
  message(FATAL_ERROR "Unsupported shared editor source API")
endif()
  if(NOT EXISTS "${CYCODE_EDITOR_ROOT}/editor/cyui-source.lock.json")
    message(FATAL_ERROR "This shared editor predates modular CyUI. Use app/editor-source.lock.json or a current --editor-root checkout.")
  endif()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/cyui-source.lock.json" CYCODE_CYUI_LOCK)
  string(JSON CYCODE_CYUI_REVISION GET "${CYCODE_CYUI_LOCK}" revision)
  string(SUBSTRING "${CYCODE_CYUI_REVISION}" 0 12 CYCODE_CYUI_CACHE_ID)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/cyui_dependency.py"
    --destination "${CYCODE_DEPS}/cyui-${CYCODE_CYUI_CACHE_ID}" --source "${CYUI_DIR}"
    OUTPUT_VARIABLE CYUI_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  # The host pins and validates CyUI above. The shared editor consumes that
  # already-validated source instead of resolving a second standalone dependency.
  # No dependency source/lock is patched, and no dirty checkout is accepted.
  set(CYCODE_CYUI_DEVELOPMENT ON)
  set(CYUI_MODULES "CodeEditor;Diff;Commands;Diagnostics;DocumentViews;Inspector;Images;Assets" CACHE STRING "CyCode UI modules" FORCE)
  set(CYCODE_EDITOR_UI_MODULES "${CYUI_MODULES}")

add_subdirectory("${CYCODE_EDITOR_ROOT}/editor" "${CMAKE_CURRENT_BINARY_DIR}/editor")
  foreach(feature ANIM_UTILS CWEBP DWEBP GIF2WEBP IMG2WEBP VWEBP WEBPINFO LIBWEBPMUX WEBPMUX EXTRAS WEBP_JS FUZZTEST)
    set(WEBP_BUILD_${feature} OFF CACHE BOOL "" FORCE)
  endforeach()
  set(WEBP_LINK_STATIC ON CACHE BOOL "" FORCE)
  add_subdirectory("${CYCODE_DEPS}/webp" "${CMAKE_BINARY_DIR}/webp" EXCLUDE_FROM_ALL)
  add_library(cycode_images STATIC src/ImageFile.cpp)
  target_include_directories(cycode_images PUBLIC include PRIVATE "${CYCODE_DEPS}/stb" "${CYCODE_DEPS}/webp/src")
  target_link_libraries(cycode_images PUBLIC cycode_files Threads::Threads PRIVATE webpdecoder)
  add_library(cycode_ui STATIC src/Application.cpp src/CommandUI.cpp src/SemanticUI.cpp src/InformationUI.cpp src/CodeActionsUI.cpp src/TerminalUI.cpp src/DebugUI.cpp src/VcsUI.cpp src/MsBuildUI.cpp src/ProjectTasksUI.cpp src/HighlightUI.cpp src/ModuleModels.cpp)
  target_include_directories(cycode_ui PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/help-routing")
  configure_file("${CYCODE_CORE_ROOT}/app/src/HelpTopics.h" "${CMAKE_CURRENT_BINARY_DIR}/help-routing/HelpTopics.h" COPYONLY)
  target_sources(cycode_ui PRIVATE src/SnippetsUI.cpp)
  target_sources(cycode_ui PRIVATE src/UserSnippetsUI.cpp)
  target_sources(cycode_ui PRIVATE src/ModuleUI.cpp src/ImageView.cpp src/ImageTextureGL.cpp src/ImagesUI.cpp)
  target_sources(cycode_ui PRIVATE src/ExtensionsUI.cpp)
  target_sources(cycode_ui PRIVATE src/PluginPackagesUI.cpp)
  target_sources(cycode_ui PRIVATE src/AdvancedDebugUI.cpp)
  target_sources(cycode_ui PRIVATE src/DebugMutationsUI.cpp)
  target_sources(cycode_ui PRIVATE src/DebugExecutionUI.cpp)
  target_sources(cycode_ui PRIVATE src/InstructionBreakpointsUI.cpp)
  target_sources(cycode_ui PRIVATE src/StepInTargetsUI.cpp)
  target_sources(cycode_ui PRIVATE src/GotoTargetsUI.cpp)
  target_sources(cycode_ui PRIVATE src/RunToCursorUI.cpp)
  target_sources(cycode_ui PRIVATE src/SessionUI.cpp src/SessionViewsUI.cpp)
  target_sources(cycode_ui PRIVATE src/DocumentDraftsUI.cpp)
  target_sources(cycode_ui PRIVATE src/GitWorktreesUI.cpp)
    target_sources(cycode_ui PRIVATE src/GitSubmodulesUI.cpp)
    target_sources(cycode_ui PRIVATE src/GitSubmoduleAddUI.cpp)
    target_sources(cycode_ui PRIVATE src/GitConflictsUI.cpp)
    target_sources(cycode_ui PRIVATE src/GitCheckoutUI.cpp)
    target_sources(cycode_ui PRIVATE src/GitMergeUI.cpp)
    target_sources(cycode_ui PRIVATE src/GitReplayUI.cpp)
    target_sources(cycode_ui PRIVATE src/UnrealAutomationUI.cpp src/UnrealLaunchUI.cpp src/UnrealArtifactsUI.cpp)
  target_sources(cycode_ui PRIVATE src/HierarchyUI.cpp src/StructureUI.cpp src/NavigationUI.cpp)
  target_sources(cycode_ui PRIVATE src/CMakePresetsUI.cpp)
  target_sources(cycode_ui PRIVATE src/WorkspaceUI.cpp)
  target_sources(cycode_ui PRIVATE src/RecoveryUI.cpp)
  target_sources(cycode_ui PRIVATE src/ResourceUI.cpp)
  target_sources(cycode_ui PRIVATE src/ProjectTemplatesUI.cpp)
  target_sources(cycode_ui PRIVATE src/TestsUI.cpp)
  target_sources(cycode_ui PRIVATE src/BuildUI.cpp)
  target_sources(cycode_ui PRIVATE src/LanguageSettingsUI.cpp)
  target_sources(cycode_ui PRIVATE src/DiagnosticAnnotations.cpp src/DiagnosticsUI.cpp src/EditorTextMap.cpp src/EditorOccurrences.cpp src/OccurrencesUI.cpp)
  target_sources(cycode_ui PRIVATE src/UiFonts.cpp src/Branding.cpp src/UpdatesUI.cpp src/UpdateStorageUI.cpp src/UpdateHistoryUI.cpp)
  target_link_libraries(cycode_ui PRIVATE cycode_updates)
  target_sources(cycode_ui PRIVATE src/ExtensionSyntax.cpp)
  target_link_libraries(cycode_ui PUBLIC cycode_core cycode_editor cycode_fileicons cycode_images
    CyUI::Core CyUI::Code CyUI::CodeEditor CyUI::Diff CyUI::Commands CyUI::Diagnostics CyUI::DocumentViews CyUI::Inspector CyUI::Images CyUI::Assets)
  add_executable(CyCode src/main.cpp)
  target_link_libraries(CyCode PRIVATE cycode_ui)
  install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/resources/fonts/" DESTINATION app/resources/fonts)
  option(CYCODE_BUILD_DOCUMENTATION "Generate and distribute the local CyDoc manual" ON)
  if(CYCODE_BUILD_DOCUMENTATION)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_custom_target(CyCode_documentation
      COMMAND "${Python3_EXECUTABLE}" "${CYCODE_CORE_ROOT}/app/.CyDocBuild/build.py"
        --ide-root "${CYCODE_ROOT}" --output "${CMAKE_BINARY_DIR}/Documentation" --output "$<TARGET_FILE_DIR:CyCode>/app/Documentation"
      WORKING_DIRECTORY "${CYCODE_CORE_ROOT}"
      COMMENT "Compiling the offline CyDoc manual" VERBATIM)
    add_dependencies(cycode_ui CyCode_documentation)
    target_include_directories(cycode_ui PRIVATE "${CYCODE_CORE_ROOT}/app/.CyDocBuild/node_modules/cydoc/examples/integrations")
    target_compile_definitions(cycode_ui PRIVATE CYCODE_LOCAL_HELP)
    if(WIN32)
      target_link_libraries(cycode_ui PRIVATE shlwapi)
    endif()
    install(DIRECTORY "${CMAKE_BINARY_DIR}/Documentation/" DESTINATION app/Documentation)
  endif()

  if(BUILD_TESTING)
    add_executable(cycode_workflow_ui_tests tests/workflow_ui_tests.cpp tests/session_ui_tests.cpp)
    target_link_libraries(cycode_workflow_ui_tests PRIVATE cycode_ui)
    add_test(NAME workflows_ui COMMAND cycode_workflow_ui_tests "${CYCODE_ROOT}" "${CMAKE_COMMAND}")
    set_tests_properties(workflows_ui PROPERTIES TIMEOUT 90 ENVIRONMENT "CYCODE_CTEST=${CMAKE_CTEST_COMMAND}")
    add_executable(cycode_template_ui_tests tests/template_ui_tests.cpp)
    target_link_libraries(cycode_template_ui_tests PRIVATE cycode_ui)
    add_dependencies(cycode_template_ui_tests cycode_extension_fixture)
    add_test(NAME templates_ui COMMAND cycode_template_ui_tests "${CYCODE_ROOT}" "$<TARGET_FILE:cycode_extension_fixture>")
    set_tests_properties(templates_ui PROPERTIES TIMEOUT 60)
    add_executable(cycode_module_tests tests/module_tests.cpp tests/problem_navigation_tests.cpp tests/outline_model_tests.cpp)
    target_link_libraries(cycode_module_tests PRIVATE cycode_ui)
    add_test(NAME ui_modules COMMAND cycode_module_tests "${CYCODE_ROOT}")
    set_tests_properties(ui_modules PROPERTIES TIMEOUT 60)
    add_executable(cycode_image_tests tests/image_tests.cpp)
    target_link_libraries(cycode_image_tests PRIVATE cycode_ui)
    add_test(NAME images COMMAND cycode_image_tests "${CYCODE_ROOT}")
    set_tests_properties(images PROPERTIES TIMEOUT 60)
    add_executable(cycode_extension_ui_tests tests/extension_ui_tests.cpp)
    target_link_libraries(cycode_extension_ui_tests PRIVATE cycode_ui)
    add_dependencies(cycode_extension_ui_tests cycode_extension_fixture)
    add_test(NAME extension_ui COMMAND cycode_extension_ui_tests "${CYCODE_ROOT}" "$<TARGET_FILE:cycode_extension_fixture>")
    set_tests_properties(extension_ui PROPERTIES TIMEOUT 45)
    add_executable(cycode_recovery_ui_tests tests/recovery_ui_tests.cpp)
    target_link_libraries(cycode_recovery_ui_tests PRIVATE cycode_ui)
    add_test(NAME recovery_ui COMMAND cycode_recovery_ui_tests "${CYCODE_ROOT}")
    set_tests_properties(recovery_ui PROPERTIES TIMEOUT 60)
    add_executable(cycode_resource_ui_tests tests/resource_ui_tests.cpp)
    target_link_libraries(cycode_resource_ui_tests PRIVATE cycode_ui)
    add_test(NAME resources_ui COMMAND cycode_resource_ui_tests "${CYCODE_ROOT}")
    set_tests_properties(resources_ui PROPERTIES TIMEOUT 60)
  endif()
  if(WIN32)
    set_target_properties(CyCode PROPERTIES WIN32_EXECUTABLE TRUE)
    target_link_options(CyCode PRIVATE /ENTRY:mainCRTStartup)
    target_link_options(CyCode PRIVATE "/MAP:${CMAKE_BINARY_DIR}/CyCode.map")
    target_link_libraries(CyCode PRIVATE shell32)
  endif()
  add_dependencies(CyCode cycode_sample cycode_unreal cycode_svn cycode_perforce cycode_mercurial cycode_diversion)

if(WIN32)
  configure_file(resources/Version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/CyCodeVersion.rc" @ONLY)
  target_sources(CyCode PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/CyCodeVersion.rc")
  add_executable(CyCodeUpdater WIN32 src/updater_main.cpp src/UpdaterRecoveryUI.cpp "${CMAKE_CURRENT_BINARY_DIR}/CyCodeVersion.rc")

  target_sources(CyCodeUpdater PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/CyCodeUpdaterManifest.rc")
  if(MSVC)
    target_link_options(CyCodeUpdater PRIVATE /MANIFEST:NO)

  target_link_libraries(CyCodeUpdater PRIVATE cycode_updates shell32 comctl32 Threads::Threads)
  target_compile_definitions(CyCodeUpdater PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN CYCODE_VERSION="${PROJECT_VERSION}")
  add_dependencies(CyCode CyCodeUpdater)
  configure_file(resources/Icons.rc.in "${CMAKE_CURRENT_BINARY_DIR}/CyCodeIcons.rc" @ONLY)
  target_sources(CyCode PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/CyCodeIcons.rc")

# Metadata/SDK facade are generated from the same Core; never fork contracts.
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_custom_command(TARGET CyCode POST_BUILD
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/stage_runtime.py"
    --core "${CYCODE_CORE_ROOT}" --ide "${CYCODE_ROOT}" --destination "$<TARGET_FILE_DIR:CyCode>"
    --cli "$<TARGET_FILE:cycode_cli>"
  VERBATIM)
add_dependencies(CyCode cycode_cli)
set_target_properties(CyCode PROPERTIES OUTPUT_NAME CyCodeIDE)
